cmake_minimum_required(VERSION 3.16)
if(CMAKE_HOST_APPLE AND NOT CMAKE_TOOLCHAIN_FILE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "10.15" CACHE STRING "Minimum macOS version")
endif()
project(wui LANGUAGES CXX C)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

if(APPLE)
    enable_language(OBJCXX)
endif()
option(WUI_BUILD_EXAMPLES "Build the WUI examples" ON)
option(WUI_BUILD_TESTS "Build platform integration tests" OFF)
if(WUI_BUILD_TESTS AND NOT EMSCRIPTEN)
    enable_testing()
    add_executable(wui_rounded_rect_test tests/common/rounded_rect.cpp)
    target_include_directories(wui_rounded_rect_test PRIVATE include)
    add_test(NAME wui_rounded_rect_test COMMAND wui_rounded_rect_test)
endif()
if(EMSCRIPTEN)
    include(cmake/WuiWasm.cmake)
endif()
add_subdirectory(src)
if(WUI_BUILD_EXAMPLES)
    add_subdirectory(examples/hello_world)
    add_subdirectory(examples/simple)
    add_subdirectory(examples/demo)
    if(EMSCRIPTEN)
        wui_wasm_site()
    endif()
endif()
if(WUI_BUILD_TESTS AND APPLE)
    enable_testing()
    add_subdirectory(tests/macos)
endif()

if(WUI_BUILD_TESTS AND EMSCRIPTEN)
    add_subdirectory(tests/wasm)
endif()
